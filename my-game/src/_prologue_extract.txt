// COMPONENT · VNBox (비주얼 노벨 연출 박스)
// ─────────────────────────────────────────
function VNBox({ line, onNext, shakeActive }) {
  const [shown, setShown] = useState('');
  const [done, setDone] = useState(false);
  const fullText = useMemo(() =>
    typeof line?.text === 'string' ? line.text : String(line?.text ?? ''),
    [line]
  );
  const imp = !!line?.important;
  const intervalRef = useRef(null);

  useEffect(() => {
    setShown('');
    setDone(false);
    if (!fullText) { setDone(true); return; }
    let i = 0;
    // 중요 대사는 좀 더 느리게 타이핑
    const speed = imp ? 42 : 28;
    intervalRef.current = setInterval(() => {
      i++;
      setShown(fullText.slice(0, i));
      if (i >= fullText.length) {
        setDone(true);
        clearInterval(intervalRef.current);
      }
    }, speed);
    return () => clearInterval(intervalRef.current);
  }, [fullText, imp]);

  const handleTap = useCallback(() => {
    if (!done) {
      // 전체 텍스트 즉시 표시
      clearInterval(intervalRef.current);
      setShown(fullText);
      setDone(true);
    } else {
      onNext?.();
    }
  }, [done, fullText, onNext]);

  const char = CHARS['kim'];

  return (
    <div
      onClick={handleTap}
      style={{
        position: 'absolute', bottom: 0, left: 0, right: 0,
        background: 'linear-gradient(transparent 0%, rgba(6,12,20,0.96) 20%)',
        padding: '64px 14px 14px', cursor: 'pointer',
        animation: 'vnUp 0.3s cubic-bezier(0.22,1,0.36,1) both',
        // 중요 대사 시 쉐이크 애니메이션
        ...(shakeActive ? { animation: 'vnUp 0.3s cubic-bezier(0.22,1,0.36,1) both, shake 0.4s ease-in-out' } : {}),
      }}
    >
      <div style={{
        background: imp
          ? 'linear-gradient(135deg,#120c06,#0f1820)'
          : 'linear-gradient(135deg,#0c1724,#091320)',
        border: `1px solid ${imp ? '#7a3a14' : '#1a3850'}`,
        borderLeft: `3px solid ${imp ? '#e0742a' : '#2e6aaa'}`,
        borderRadius: 10, padding: '14px 16px 12px',
        boxShadow: imp
          ? '0 4px 28px rgba(220,80,30,0.22), inset 0 0 40px rgba(200,60,20,0.04)'
          : '0 4px 20px rgba(0,0,0,0.55)',
        // 중요 대사 시 붉은 틴트 테두리
        outline: imp ? '1px solid rgba(200,60,20,0.12)' : 'none',
        outlineOffset: 3,
      }}>
        <div style={{
          fontSize: 11.5, fontWeight: 700,
          color: imp ? '#d08030' : char.accentColor,
          marginBottom: 9,
          fontFamily: 'system-ui, sans-serif',
          letterSpacing: '0.05em',
          display: 'flex', alignItems: 'center', gap: 6,
        }}>
          {imp && <span style={{ opacity: 0.8, fontSize: 9 }}>⚠</span>}
          {char.name}
          <span style={{ fontSize: 9.5, opacity: 0.45, fontWeight: 400, marginLeft: 2 }}>{char.dept}</span>
        </div>
        <div style={{
          fontSize: 15.5, lineHeight: 1.72,
          color: imp ? '#f2c890' : '#bcd8f2',
          fontFamily: "'Noto Sans KR', system-ui, sans-serif",
          minHeight: 28,
          wordBreak: 'keep-all', overflowWrap: 'break-word',
        }}>
          {shown}
          {/* 타이핑 커서 */}
          {!done && (
            <span style={{
              borderRight: `2px solid ${imp ? '#e0742a' : '#4a8ac0'}`,
              animation: 'blink 0.5s step-end infinite',
              marginLeft: 1,
            }}>&nbsp;</span>
          )}
        </div>
        {done && (
          <div style={{
            textAlign: 'right', marginTop: 10, fontSize: 10.5,
            color: imp ? '#905020' : '#2a5a7a',
            fontFamily: 'monospace', letterSpacing: '0.12em',
            animation: 'blink 1.4s ease-in-out infinite',
          }}>
            ▶ 탭하여 계속
          </div>
        )}
      </div>
    </div>
  );
}

// ─────────────────────────────────────────
// SCREEN · Messenger (메인 채팅 화면)
// ─────────────────────────────────────────
function MessengerScreen({ onEnd }) {
  const [messages, setMessages]       = useState([]);
  const [emotion, setEmotion]         = useState('neutral');
  const [isTyping, setIsTyping]       = useState(false);
  const [nickname, setNickname]       = useState(null);

  // 선택지 / 입력 상태
  const [pendingChoices, setPendingChoices] = useState(null); // null | { choices }
  const [awaitInput, setAwaitInput]         = useState(false);
  const [choicesLocked, setChoicesLocked]   = useState(false);

  // VN 상태
  const [vnState, setVnState] = useState(null); // null | { lines, idx, important, autoNext }

  // 화면 쉐이크 상태 (중요 VN 대사 시)
  const [shaking, setShaking]   = useState(false);
  const [redTint, setRedTint]   = useState(false);

  const scrollRef  = useRef(null);
  const timersRef  = useRef([]);   // 정리할 타이머 목록
  const busyRef    = useRef(false);
  const vnLockRef  = useRef(false);
  const nicknameRef = useRef(null); // 닉네임 최신값 동기 접근용

  // 닉네임을 ref에도 동기화
  useEffect(() => { nicknameRef.current = nickname; }, [nickname]);

  // 언마운트 시 모든 타이머 정리
  useEffect(() => () => { timersRef.current.forEach(clearTimeout); }, []);

  // ── 안전한 타이머 등록 ──
  const later = useCallback((fn, ms) => {
    const t = setTimeout(fn, ms);
    timersRef.current.push(t);
    return t;
  }, []);

  // ── 스크롤 하단으로 ──
  const scrollDown = useCallback(() => {
    requestAnimationFrame(() => {
      if (scrollRef.current) {
        scrollRef.current.scrollTo({ top: scrollRef.current.scrollHeight, behavior: 'smooth' });
      }
    });
  }, []);

  // ── 메시지 추가 (stable key 보장) ──
  const addMsg = useCallback((msg) => {
    setMessages(prev => [...prev, { ...msg, id: uid() }]);
    setTimeout(scrollDown, 60);
  }, [scrollDown]);

  // ── 라인 해석 (함수형 라인 지원) ──
  const resolveLine = useCallback((line) => {
    if (typeof line === 'function') {
      return line({ nickname: nicknameRef.current ?? '플레이어' });
    }
    return String(line ?? '');
  }, []);

  // ── 화면 쉐이크 + 붉은 틴트 트리거 ──
  const triggerShake = useCallback(() => {
    setShaking(true);
    setRedTint(true);
    later(() => setShaking(false), 500);
    later(() => setRedTint(false), 800);
  }, [later]);

  // ─────────────────────
  // 핵심: 노드 실행 함수
  // ─────────────────────
  const runNode = useCallback((nodeId) => {
    const node = dialogueScript[nodeId];
    if (!node) {
      console.warn('[TalkLine] 알 수 없는 노드:', nodeId);
      return;
    }

    // 이전 상태 초기화
    busyRef.current = true;
    setPendingChoices(null);
    setAwaitInput(false);
    setChoicesLocked(false);

    // 감정 업데이트
    if (node.emotion) setEmotion(node.emotion);

    // ── VN 모드 ──
    if (node.isVn) {
      const lines = (node.lines || []).map(resolveLine);
      setVnState({
        lines,
        idx: 0,
        important: !!node.important,
        autoNext: node.autoNext ?? null,
        nodeId,
      });
      // 중요 VN 대사: 쉐이크 + 틴트
      if (node.important) triggerShake();
      busyRef.current = false;
      return;
    }

    // ── 채팅 모드 ──
    const lines = (node.lines || []).map(resolveLine);
    let delay = 0;

    // 각 라인 타이핑 → 버블 순차 출력
    lines.forEach((text, idx) => {
      // 타이핑 속도: 기본 글자 수 × 20ms (최소 400ms)
      const typingMs = Math.max(400, text.length * 20);

      later(() => {
        setIsTyping(true);
        scrollDown();
      }, delay);
      delay += typingMs;

      later(() => {
        setIsTyping(false);
        addMsg({ type: 'recv', char: 'kim', text, showName: idx === 0 });
      }, delay);
      delay += 100;
    });

    // 모든 라인 출력 후 처리
    later(() => {
      busyRef.current = false;

      // 시스템 메시지
      if (node.sysMsg) {
